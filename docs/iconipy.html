iconipy

iconipy

Say goodbye to the tedious hassle of graphic programs! Now you can create stunning icons directly from your code. With the look and feel defined right in the code, adjustments are a breeze. Plus, the included icon sets can easily be expanded with your own sets based on font files. 🎨✨

Installation

pip install iconipy

Usage

First you initialize an "IconFactory" with an icon set and look-and-feel settings like this:

from iconipy import IconFactory 

create_button_icon = IconFactory(
                        icon_set = 'lucide', 
                        icon_size = (128,64), # Providing just an integer would produce a square icon 
                        font_size = 38,  
                        font_color = (0, 0, 0, 255), # Solid black in RGBA format (R, G, B, A)
                        outline_color = 'dimgrey',   # Color names are also supported
                        outline_width = 6,
                        background_color = 'silver', 
                        background_radius = 10
)

Then you create your icons: 

icon_home = create_button_icon.asPil('house') # PIL Image
icon_reload = create_button_icon.asQPixmap('refresh-cw') # Qt based frameworks
icon_files = create_button_icon.asTkPhotoImage('files') # tkinter, ttkbootstrap, PySimpleGUI, FreeSimpleGUI
icon_exit_app = create_button_icon.asRawList('log-out') # DearPyGUI

Depending on your GUI toolkit's whims, you can create PIL Image Objects, ByteIO Objects, Byte-Strings, Raw Pixel Lists, TkPhotoImage Objects, QImage Objects, save to file, and more.

You need to preview an icon? Here we go:

create_button_icon.show('house')

The icon sets you can choose from include: lucide, boxicons, ligature_symbols, lineicons, material_icons_regular, material_icons_round_regular, material_icons_sharp_regular, material_icons_outlined_regular, microns and typicons.

Feeling adventurous? Dump all the icons to your hard drive and explore:

create_button_icon.saveAll(<path to target directory>)

You need a hand? Check what the icon set has to offer:

print(create_button_icon.search('hand'))

Just want a list with all icon names? No problem:

print(create_button_icon.icon_names)

More info

Visit https://github.com/digidigital/iconipy or https://iconipy.digidigital.de for sample code for the most popular GUI toolkits and detailed documentation.

394    def _read_codepoints(self):
395        codepoints = {}
396        for icon_set in _ICON_SETS.keys():
397            icon_set_codepoints = {}
398            _METADATA_FILE = _ICON_SETS[icon_set]["METADATA_FILE"]
399
400            if icon_set == "lucide":
401                with open(_METADATA_FILE) as json_data:
402                    codepoint_data = json.load(json_data)
403                for key in codepoint_data.keys():
404                    icon_set_codepoints[key] = codepoint_data[key]["encodedCode"][1:]
405
406            elif icon_set in ("boxicons", "microns", "typicons"):
407                pattern = re.compile(
408                    r'.*\.(?P<codepoint_key>([a-z0-9]*-){1,4}[a-z0-9]*):.*\n.*"\\(?P<codepoint_value>.*)";'
409                )
410
411                with open(_METADATA_FILE, "r") as boxicons_codepoint_file:
412                    raw_content = boxicons_codepoint_file.read()
413
414                    # Find all matches in the file content
415                    matches = pattern.finditer(raw_content)
416
417                    # Populate the dictionary with the matches
418                    for match in matches:
419                        codepoint_key = match.group("codepoint_key")
420                        codepoint_value = match.group("codepoint_value")
421                        icon_set_codepoints[codepoint_key] = codepoint_value
422
423            elif icon_set == "ligature_symbols":
424                # Read the HTML content from a file
425                with open(_METADATA_FILE, 'r', encoding='utf-8') as ls_codepoint_file:
426                    html_string = ls_codepoint_file.read()
427
428                pattern = re.compile(
429                    r'<td class="lsf symbol">(.+?)</td>\s*<td class="ligature">.+?</td>\s*<td class="unicode">\\(.+?)</td>'
430                )
431                
432                # Find matches
433                matches = pattern.findall(html_string)
434                
435                # Populate the dictionary with the matches
436                icon_set_codepoints = {match[0]: match[1] for match in matches}              
437                
438            elif icon_set == "lineicons":
439                with open(_METADATA_FILE) as json_data:
440                    codepoint_data = json.load(json_data)
441                for key, value in codepoint_data.items():
442                    icon_set_codepoints[key] = hex(value)[2:]
443
444            else:
445                # Material icons
446                with open(_METADATA_FILE) as material_icons_codepoint_file:
447                    for line in material_icons_codepoint_file:
448                        codepoint_key, codepoint_value = line.strip().split()
449                        icon_set_codepoints[codepoint_key] = codepoint_value
450
451            codepoints[icon_set] = icon_set_codepoints
452
453        return codepoints

463    def _draw_character(
464        self,
465        character,
466        font_path,
467        font_size=32,
468        font_color="grey",
469        icon_size=32,
470        icon_background_color=None,
471        icon_outline_width=0,
472        icon_background_radius=0,
473        icon_outline_color="dimgrey",
474    ):
475
476        # Create image
477        if isinstance (icon_size, int):
478            width = height = icon_size
479        elif isinstance (icon_size, tuple) and isinstance (icon_size[0], int) and isinstance (icon_size[1], int):
480            width = icon_size[0]
481            height = icon_size[1]
482        else:
483            raise AttributeError ('icon_size must be of type int or tuple (int, int)')

486        if icon_background_color or icon_outline_width:

495            image = Image.new("RGBA", (width, height), (0, 0, 0, 0))

498            # Load font
499            font = ImageFont.truetype(font_path, font_size)
500
501            # Draw character
502            draw = ImageDraw.Draw(image)
503
504            x = width // 2
505            y = height // 2
506
507            draw.text((x, y), character, font=font, anchor="mm", fill=font_color)

510    def _image_round_background(
511        self,
512        size = (64,64),
513        fill = "silver",
514        outline = "grey",
515        outline_width = 7,
516        outline_radius = 10,
517        factor = 3,
518    ):
519        """Create and return a background image with rounded corners. Set the outline radius to size/2 to achieve a circular background."""
520        width = size[0] 
521        height = size[1]
522        im = Image.new("RGBA", (factor * width, factor * height), (0, 0, 0, 0))
523        draw = ImageDraw.Draw(im, "RGBA")
524        draw.rounded_rectangle(
525            (0, 0, (factor * width)-1, (factor * height)-1),
526            radius=factor * outline_radius,
527            outline=outline,
528            fill=fill,
529            width=factor * outline_width,
530        )
531        im = im.resize((width, height), Image.LANCZOS)
532        return im
